uint8_t taskCount = 0;
int8_t currentTask = -1;

void taskReturnHandler(int returnValue) {
    taskList[currentTask].returnValue = returnValue;
    taskList[currentTask].state = TASK_FINISHED;
    SysTick_Handler();
}

void addTask(void (*taskFunction)(void *), void *param) {
    if (taskCount < MAX_TASKS) {
        initTaskStack(&taskList[taskCount], taskFunction, param);
        taskCount++;
		// 璋冭瘯鎵撳嵃
        printf("Task added. Task count: %d\n", taskCount);
    }
}
