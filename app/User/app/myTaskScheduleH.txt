#ifndef MY_TASK_SCHEDULE_H
#define MY_TASK_SCHEDULE_H

#define STACK_SIZE 128
#define MAX_TASKS 10

typedef enum {
    TASK_READY,
    TASK_RUNNING,
    TASK_WAITING,
    TASK_FINISHED
} TaskState;

typedef struct {
    void (*taskFunction)(void *param);
    void *param;
    TaskState state;
    uint32_t *stackPointer;
    uint32_t stack[STACK_SIZE];
    int returnValue;
	  unsigned char priority;
} Task;


void initTaskStack(Task *task, void (*taskFunction)(void *), void *param);
void addTask(void (*taskFunction)(void *), void *param);
__asm void saveContext(uint32_t **stackPointer); 
__asm void restoreContext(uint32_t **stackPointer);

extern Task taskList[MAX_TASKS];
extern uint8_t taskCount;
extern int8_t currentTask;

#endif
